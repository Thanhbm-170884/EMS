create database hrms_db;

use hrms_db;

create table accounts
(
    Id           int auto_increment
        primary key,
    Username     varchar(50)      not null,
    PasswordHash varchar(255)     not null,
    Status       bit default b'1' not null,
    UserId       int              not null,
    constraint UserId
        unique (UserId),
    constraint Username
        unique (Username)
);

create table departments
(
    Id            int auto_increment
        primary key,
    Code          varchar(20)                  not null,
    Name          varchar(100) charset utf8mb3 not null,
    HeadAccountId int                          null,
    constraint Code
        unique (Code),
    constraint FK_Departments_Accounts
        foreign key (HeadAccountId) references accounts (Id)
);

CREATE TABLE holidaytemplates
(
    Id                   INT AUTO_INCREMENT PRIMARY KEY,
    HolidayName          VARCHAR(100) CHARSET utf8mb3 NOT NULL,
    RecurType            ENUM('FIXED_SOLAR', 'LUNAR') NOT NULL,
    FixedMonth           TINYINT NULL,              -- chỉ dùng khi FIXED_SOLAR
    FixedDay             TINYINT NULL,              -- chỉ dùng khi FIXED_SOLAR
    FixedDurationDays     TINYINT DEFAULT 1 NOT NULL,-- số ngày nghỉ liên tiếp
    DefaultCoefficient   DECIMAL(3,1) DEFAULT 1.0 NOT NULL,
    IsCoefficientLocked  BIT DEFAULT b'0' NOT NULL, -- checkbox "Cố định hệ số giữa các năm"
    IsActive             BIT DEFAULT b'1' NOT NULL,
    CreatedBy            INT NULL,
    CONSTRAINT FK_HolidayTemplates_Accounts
        FOREIGN KEY (CreatedBy) REFERENCES accounts (Id)
);

CREATE TABLE holidayyearinstances
(
    Id           INT AUTO_INCREMENT PRIMARY KEY,
    TemplateId   INT NOT NULL,
    Year         INT NOT NULL,
    StartDate    DATE NOT NULL,
    EndDate      DATE NOT NULL,
    Coefficient  DECIMAL(3,1) NOT NULL,
    CreatedBy    INT NULL,
    CreatedAt    DATETIME DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT UQ_HYI_Template_Year UNIQUE (TemplateId, Year),
    CONSTRAINT FK_HYI_Template
        FOREIGN KEY (TemplateId) REFERENCES holidaytemplates (Id) ON DELETE CASCADE,
    CONSTRAINT FK_HYI_Accounts
        FOREIGN KEY (CreatedBy) REFERENCES accounts (Id),
    CONSTRAINT CHK_HYI_DateRange CHECK (StartDate <= EndDate)
);

create table requesttypes
(
    Id   int auto_increment

    constraint Code

create table requests
(
    Id                       int auto_increment
        primary key,

RejectionReason VARCHAR(255) NULL,

create table roles
(

    constraint Name

create table accountroles
(
    AccountId int not null,

create table shifts

create table positions

CREATE TABLE payrollconfigs (
    Id                      INT AUTO_INCREMENT PRIMARY KEY,
    ConfigName              VARCHAR(100) NOT NULL,
    EffectiveDate           DATE NOT NULL,
    BhxhPercent             DECIMAL(5, 2) DEFAULT 8.00 NOT NULL,  -- 8% BHXH
    BhytPercent             DECIMAL(5, 2) DEFAULT 1.50 NOT NULL,  -- 1.5% BHYT
    BhtnPercent             DECIMAL(5, 2) DEFAULT 1.00 NOT NULL,  -- 1% BHTN
    MaxInsuranceSalary      DECIMAL(18, 2) NULL,                  -- Lương đóng BH trần
    PersonalTaxDeduction    DECIMAL(18, 2) DEFAULT 11000000.00 NOT NULL,
    DependentTaxDeduction   DECIMAL(18, 2) DEFAULT 4400000.00 NOT NULL,
    StandardWorkingDays     INT DEFAULT 22 NOT NULL,              -- Số ngày công chuẩn trong tháng
    OtWeekdayRate           DECIMAL(3, 2) DEFAULT 1.50 NOT NULL,  -- Hệ số OT ngày thường (150%)
    OtWeekendRate           DECIMAL(3, 2) DEFAULT 2.00 NOT NULL,  -- Hệ số OT cuối tuần (200%)
    OtHolidayRate           DECIMAL(3, 2) DEFAULT 3.00 NOT NULL,  -- Hệ số OT ngày lễ (300%)
    IsActive                BIT DEFAULT b'1' NOT NULL,

    CreatedAt               DATETIME DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT FK_PayrollConfigs_Accounts FOREIGN KEY (CreatedByAccountId) REFERENCES accounts (Id)
);

CREATE TABLE allowancetypes (
    Id                  INT AUTO_INCREMENT PRIMARY KEY,
    Code                VARCHAR(20) NOT NULL,                 -- VD: PC_AN_TRUA, PC_XANG_XE
    Name                VARCHAR(100) CHARSET utf8mb3 NOT NULL,
    Type                VARCHAR(20) DEFAULT 'Allowance' NOT NULL, -- 'Allowance' (Phụ cấp cộng thêm) hoặc 'Deduction' (Giảm trừ)
    CalculationMethod   VARCHAR(30) DEFAULT 'Fixed' NOT NULL, -- 'Fixed' (Cố định tháng) hoặc 'ByWorkDay' (Theo ngày đi làm)
    DefaultAmount       DECIMAL(18, 2) DEFAULT 0.00 NOT NULL,
    IsTaxable           BIT DEFAULT b'1' NOT NULL,            -- Có tính thuế TNCN không
    TaxExemptLimit      DECIMAL(18, 2) DEFAULT 0.00 NOT NULL, -- Hạn mức miễn thuế 
    IsInsuranceSalary   BIT DEFAULT b'0' NOT NULL,            -- Có tính vào lương đóng BHXH không
    IsActive            BIT DEFAULT b'1' NOT NULL,
    CONSTRAINT UQ_AllowanceTypes_Code UNIQUE (Code)
);


CREATE TABLE payslips
(
    Id                      INT AUTO_INCREMENT PRIMARY KEY,
    UserId                  INT NOT NULL,           
    PeriodId                INT NOT NULL,     
    
    -- 1. Ngày công & Giờ làm việc
    StandardWorkDays        INT DEFAULT 22 NOT NULL, 
    ActualWorkDays          DECIMAL(4, 1) DEFAULT 0.0 NOT NULL,
    OtHours                 DECIMAL(5, 2) DEFAULT 0.00 NOT NULL,
    
    -- 2. Lương cơ bản & Lương theo ngày công
    BaseSalary              DECIMAL(18, 2) NOT NULL,
    ActualBaseSalary        DECIMAL(18, 2) DEFAULT 0.00 NOT NULL,      -- Lương thực tế = (BaseSalary / Standard) * Actual
    OtSalary                DECIMAL(18, 2) DEFAULT 0.00 NOT NULL,
    
    -- 3. PHỤ CẤP
    TotalAllowanceAmount DECIMAL(18, 2) DEFAULT 0.00 NOT NULL,
    
    -- 4. Thưởng & Tổng thu nhập Gross
    BonusAmount             DECIMAL(18, 2) DEFAULT 0.00 NOT NULL,
    GrossAmount             DECIMAL(18, 2) NOT NULL,                   -- Tổng Gross = ActualBaseSalary + OtSalary + 3 Phụ cấp + Bonus
    
    -- 5. Các khoản giảm trừ & Thuế TNCN
    BhxhAmount DECIMAL(18, 2) DEFAULT 0.00 NOT NULL, -- BHXH (8.0%)
    BhytAmount DECIMAL(18, 2) DEFAULT 0.00 NOT NULL, -- BHYT (1.5%)
    BhtnAmount DECIMAL(18, 2) DEFAULT 0.00 NOT NULL, -- BHTN (1.0%)
    TotalInsuranceDeduction DECIMAL(18, 2) DEFAULT 0.00 NOT NULL, -- Tổng trừ BH = Bhxh + Bhyt + Bhtn

    DependentsCount         INT DEFAULT 0 NOT NULL,                    -- Số người phụ thuộc tại thời điểm tính
    DependentDeduction      DECIMAL(18, 2) DEFAULT 0.00 NOT NULL,      -- Mức giảm trừ gia cảnh (11Tr + Người PT * 4.4Tr)
    TaxableIncome           DECIMAL(18, 2) DEFAULT 0.00 NOT NULL,      -- Thu nhập tính thuế
    TaxDeduction            DECIMAL(18, 2) DEFAULT 0.00 NOT NULL,      -- Tiền thuế TNCN khấu trừ
    PenaltyAmount           DECIMAL(18, 2) DEFAULT 0.00 NOT NULL,      -- Tiền phạt vi phạm
    AdvanceAmount           DECIMAL(18, 2) DEFAULT 0.00 NOT NULL,      -- Tạm ứng trừ lại
    OtherDeductions         DECIMAL(18, 2) DEFAULT 0.00 NOT NULL,      -- Khấu trừ khác
    
    -- 6. Thực lĩnh & Trạng thái
    NetAmount               DECIMAL(18, 2) NOT NULL,
    Status                  VARCHAR(20) DEFAULT 'Draft' NOT NULL,      -- 'Draft', 'Confirmed', 'Paid'
    Note                    VARCHAR(255) NULL,
    
    AdjustedByAccountId     INT NULL,
    CreatedAt               DATETIME DEFAULT CURRENT_TIMESTAMP NOT NULL,
    UpdatedAt               DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP NOT NULL,
    
    CONSTRAINT UQ_Payslips_User_Period UNIQUE (UserId, PeriodId),
    CONSTRAINT FK_Payslips_Users FOREIGN KEY (UserId) REFERENCES users (Id) ON DELETE CASCADE,
    CONSTRAINT FK_Payslips_Periods FOREIGN KEY (PeriodId) REFERENCES timesheetperiods (Id),
    CONSTRAINT FK_Payslips_Accounts FOREIGN KEY (AdjustedByAccountId) REFERENCES accounts (Id)
);

CREATE TABLE payslip_allowances
(
    Id              INT AUTO_INCREMENT PRIMARY KEY,

    Amount          DECIMAL(18, 2) DEFAULT 0.00 NOT NULL,-- Số tiền thực nhận tháng này

    CONSTRAINT FK_PayslipAllowances_AllowanceTypes

create table shiftassignmentbatches
(
    Id                int auto_increment
        primary key,
    Name              varchar(100)                                                 not null,
    ShiftId           int                                                          not null,
    StartDate         date                                                         not null,
    EndDate           date                                                         null,
    RecurType         enum ('NONE', 'WEEKLY', 'MONTHLY') default 'WEEKLY'          not null,
    RecurInterval     tinyint                            default 1                 not null,
    MonthlyType       enum ('WEEKDAY', 'DATE')                                     null,
    MonthlyWeekday    tinyint                                                      null,
    MonthlyOccurrence tinyint                                                      null,
    MonthlyDay        tinyint                                                      null,
    CreatedBy         int                                                          null,
    CreatedAt         datetime                           default CURRENT_TIMESTAMP not null,
    constraint FK_SAB_Accounts
        foreign key (CreatedBy) references accounts (Id),
    constraint FK_SAB_Shifts
        foreign key (ShiftId) references shifts (Id),
    constraint CHK_SAB_DateRange
        check ((`EndDate` is null) or (`StartDate` <= `EndDate`))
);
